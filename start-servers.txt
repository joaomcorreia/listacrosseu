# ListAcross.eu - Server Startup Commands
# Multilingual SaaS with 27 EU languages + AI Assistant

# === POWERSHELL COMMANDS ===
# Copy these to PowerShell terminals (RECOMMENDED):

# Terminal 1 (Backend - PowerShell):
Set-Location C:\projects\listacrosseu\backend; & "C:\projects\listacrosseu\.venv\Scripts\Activate.ps1"; python manage.py runserver 0.0.0.0:8000

# Terminal 2 (Frontend - PowerShell):
Push-Location C:\projects\listacrosseu\frontend; npm run dev

# === COMMAND PROMPT (CMD) COMMANDS ===
# Copy these to Command Prompt terminals (use if PowerShell unavailable):

# Terminal 1 (Backend - CMD):
cd /d C:\projects\listacrosseu\backend && C:\projects\listacrosseu\.venv\Scripts\activate.bat && python manage.py runserver 0.0.0.0:8000

# Terminal 2 (Frontend - CMD):
cd /d C:\projects\listacrosseu\frontend && npm run dev

# IMPORTANT NOTE:
# - PowerShell uses: Set-Location, semicolon (;) for command chaining
# - CMD uses: cd /d, double ampersand (&&) for command chaining
# - Do NOT mix PowerShell and CMD syntax in the same terminal!
